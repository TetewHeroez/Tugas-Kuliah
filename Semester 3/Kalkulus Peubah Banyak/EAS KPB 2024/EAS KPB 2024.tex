        Jacobian transformasi baliknya adalah
        \[
          \frac{\partial(x,y)}{\partial(u,v)}
          =\begin{vmatrix}
            1                   & 1                  \\[2pt]
            -\dfrac{v}{(u+v)^2} & \dfrac{u}{(u+v)^2}
          \end{vmatrix}
          =\frac{u+v}{(u+v)^2}=\frac1{u+v}.
        \]
        Karena $u+v\ge2$, diperoleh $dA=\dfrac1{u+v}\,du\,dv$.
        Misalkan kerapatan luas keping adalah konstanta $\rho$.
        Jarak titik $(x,y)$ ke sumbu $y$ adalah $|x|$, sehingga
        \begin{align*}
          I_y & =\iint_D\rho x^2\,dA
          =\rho\int_1^2\int_1^3\frac{(u+v)^2}{u+v}\,dv\,du                 \\
              & =\rho\int_1^2\left[uv+\frac{v^2}{2}\right]_{v=1}^{v=3}\,du \\
              & =\rho\int_1^2(2u+4)\,du
          =\rho[u^2+4u]_1^2=\boxed{7\rho}.
        \end{align*}
        Untuk kerapatan satuan, $\rho=1$, sehingga $I_y=7$.

  \item Tuliskan permukaannya sebagai
        \[
          z=4-\frac{x^2+y^2}{2},\qquad x^2+y^2\le3.
        \]
        Permukaan ini adalah bagian paraboloida di dalam tabung berjari-jari $\sqrt3$.
        Puncaknya $(0,0,4)$ dan tepi lingkarannya berada pada $z=\frac52$.
        Misalkan permukaan ini $\Sigma$ dan kerapatan luasnya konstan $\rho$.

        Karena yang dicari adalah pusat massa permukaan, gunakan elemen luas
        \[
          dS=\sqrt{1+z_x^2+z_y^2}\,dx\,dy
          =\sqrt{1+x^2+y^2}\,dx\,dy,
        \]
        dengan $z_x=-x$ dan $z_y=-y$.
        Gunakan koordinat polar
        \[
          x=r\cos\theta,\quad y=r\sin\theta,\quad
          0\le r\le\sqrt3,\quad0\le\theta\le2\pi.
        \]
        Maka $z=4-\frac12r^2$ dan $dS=r\sqrt{1+r^2}\,dr\,d\theta$.
        Massa permukaan adalah
        \begin{align*}
          m & =\iint_\Sigma\rho\,dS
          =2\pi\rho\int_0^{\sqrt3}r\sqrt{1+r^2}\,dr                    \\
            & =\frac{2\pi\rho}{3}\left[(1+r^2)^{3/2}\right]_0^{\sqrt3}
          =\frac{2\pi\rho}{3}(8-1)=\frac{14\pi\rho}{3}.
        \end{align*}
        Simetri terhadap bidang $yz$ dan $xz$ memberikan $\bar x=\bar y=0$.
        Untuk koordinat $\bar z$, hitung momen terhadap bidang $xy$:
        \[
          M_{xy}=\iint_\Sigma\rho z\,dS
          =2\pi\rho\int_0^{\sqrt3}
          \left(4-\frac{r^2}{2}\right)r\sqrt{1+r^2}\,dr.
        \]
        Substitusikan $t=1+r^2$, sehingga $dt=2r\,dr$ dan
        $4-\frac12r^2=\frac{9-t}{2}$.
        Batas $r=0,\sqrt3$ berubah menjadi $t=1,4$.
        Dengan demikian,
        \begin{align*}
          M_{xy} & =\frac{\pi\rho}{2}\int_1^4(9-t)t^{1/2}\,dt \\
                 & =\frac{\pi\rho}{2}
          \left[6t^{3/2}-\frac25t^{5/2}\right]_1^4
          =\frac{74\pi\rho}{5}.
        \end{align*}
        Jadi, pusat massanya adalah
        \[
          \bar z=\frac{M_{xy}}m
          =\frac{74\pi\rho/5}{14\pi\rho/3}=\frac{111}{35},
          \qquad
          \boxed{(\bar x,\bar y,\bar z)=\left(0,0,\frac{111}{35}\right).}
        \]
