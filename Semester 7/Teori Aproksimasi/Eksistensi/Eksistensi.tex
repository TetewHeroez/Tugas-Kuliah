\subsection{Teorema 3.9}

\begin{frame}{\insertsection}{\insertsubsection}
  \setcounter{teorema}{8}
  \begin{teorema}
    Misalkan $\mathcal{F}$ adalah ruang Euclidean dengan hasil kali dalam $\langle \cdot, \cdot \rangle$ dan norma $\| \cdot \| = \langle \cdot, \cdot \rangle^{1/2}$. Maka identitas jajar genjang
    \begin{equation}
      \|v+w\|^2 + \|v-w\|^2 = 2\|v\|^2 + 2\|w\|^2 \quad \text{untuk semua } v,w \in \mathcal{F} \label{eq:3.1}
    \end{equation}
    berlaku. Jika $\mathcal{F}$ adalah ruang Euclidean atas bilangan real $\R$, maka identitas polarisasi
    \begin{equation}
      \langle v,w \rangle = \frac{1}{4}(\|v+w\|^2 - \|v-w\|^2) \quad \text{untuk semua } v,w \in \mathcal{F} \label{eq:3.2}
    \end{equation}
    berlaku. Jika $\mathcal{F}$ adalah ruang Euclidean atas bilangan kompleks $\C$, maka identitas polarisasi berlaku sebagai
    \begin{equation}
      \langle v,w \rangle = \frac{1}{4}(\|v+w\|^2 - \|v-w\|^2 + i\|v+iw\|^2 - i\|v-iw\|^2)\quad \text{untuk semua } v,w \in \mathcal{F}. \label{eq:3.3}
    \end{equation}
  \end{teorema}
\end{frame}

\begin{frame}[allowframebreaks]{\insertsection}{\insertsubsection}
  \textit{Bukti.}
  Persamaan \eqref{eq:3.1} dan \eqref{eq:3.2} mengikuti langsung dari sifat linearitas hasil kali dalam:
  \begin{equation*}
    \|v \pm w\|^2 = \langle v \pm w, v \pm w \rangle = \langle v,v \rangle \pm 2\langle v,w \rangle + \langle w,w \rangle = \|v\|^2 \pm 2\langle v,w \rangle + \|w\|^2.
  \end{equation*}
  Dengan menjumlahkan persamaan untuk $\|v+w\|^2$ dan $\|v-w\|^2$, kita peroleh identitas jajar genjang \eqref{eq:3.1}:
  \begin{equation*}
    \|v+w\|^2 + \|v-w\|^2 = 2\|v\|^2 + 2\|w\|^2.
  \end{equation*}
  Dengan mengurangkan $\|v-w\|^2$ dari $\|v+w\|^2$, kita peroleh identitas polarisasi untuk ruang real \eqref{eq:3.2}:
  \begin{equation*}
    \|v+w\|^2 - \|v-w\|^2 = 4\langle v,w \rangle \implies \langle v,w \rangle = \frac{1}{4}(\|v+w\|^2 - \|v-w\|^2).
  \end{equation*}
  \framebreak

  Untuk identitas polarisasi atas bilangan kompleks \eqref{eq:3.3}, kita ingat bahwa $\langle w, v \rangle = \overline{\langle v, w \rangle}$, $\langle \alpha v, w \rangle = \alpha \langle v, w \rangle$, dan $\langle v, \alpha w \rangle = \bar{\alpha} \langle v, w \rangle$. Sehingga,
  \begin{align*}
    \|v+w\|^2 & = \|v\|^2 + \langle v,w \rangle + \langle w,v \rangle + \|w\|^2, \\
    \|v-w\|^2 & = \|v\|^2 - \langle v,w \rangle - \langle w,v \rangle + \|w\|^2.
  \end{align*}
  Pengurangan keduanya memberikan:
  \begin{equation*}
    \|v+w\|^2 - \|v-w\|^2 = 2\langle v,w \rangle + 2\langle w,v \rangle.
  \end{equation*}
  Selanjutnya,
  \begin{align*}
    \|v+iw\|^2 & = \|v\|^2 - i\langle v,w \rangle + i\langle w,v \rangle + \|w\|^2, \\
    \|v-iw\|^2 & = \|v\|^2 + i\langle v,w \rangle - i\langle w,v \rangle + \|w\|^2.
  \end{align*}
  \framebreak

  Pengurangan keduanya memberikan:
  \begin{equation*}
    \|v+iw\|^2 - \|v-iw\|^2 = -2i\langle v,w \rangle + 2i\langle w,v \rangle.
  \end{equation*}
  Dengan mengalikan hasil ini dengan $i$, kita peroleh:
  \begin{equation*}
    i\|v+iw\|^2 - i\|v-iw\|^2 = 2\langle v,w \rangle - 2\langle w,v \rangle.
  \end{equation*}
  Dengan menjumlahkan bagian real dan imajinernya:
  \begin{equation*}
    \|v+w\|^2 - \|v-w\|^2 + i\|v+iw\|^2 - i\|v-iw\|^2 = 4\langle v,w \rangle.
  \end{equation*}
  Hal ini melengkapi bukti identitas polarisasi \eqref{eq:3.3}.
\end{frame}

\subsection{Teorema 3.10}

\begin{frame}[allowframebreaks]{\insertsection}{\insertsubsection}
  \begin{teorema}[Teorema Jordan-von Neumann, 1935]
    Misalkan $\mathcal{F}$ adalah ruang linear dengan norma $\| \cdot \|$, di mana identitas jajar genjang \eqref{eq:3.1} berlaku. Maka terdapat sebuah hasil kali dalam $\langle \cdot, \cdot \rangle$ pada $\mathcal{F}$, sehingga
    \begin{equation}
      \langle v,v \rangle = \|v\|^2 \quad \text{untuk semua } v \in \mathcal{F}, \label{eq:3.4}
    \end{equation}
    yaitu, $\mathcal{F}$ adalah ruang Euclidean.
  \end{teorema}

  \textit{Bukti.}
  Misalkan $\mathcal{F}$ adalah ruang linear atas $\R$. Definisikan pemetaan $\langle \cdot, \cdot \rangle : \mathcal{F} \times \mathcal{F} \to \R$ melalui identitas polarisasi \eqref{eq:3.2}:
  \begin{equation*}
    \langle v,w \rangle := \frac{1}{4}(\|v+w\|^2 - \|v-w\|^2) \quad \text{untuk } v,w \in \mathcal{F}.
  \end{equation*}
  Akan diverifikasi aksioma hasil kali dalam:
  \begin{itemize}
    \item \textbf{Definit Positif:}
          $\langle v,v \rangle = \frac{1}{4}(\|2v\|^2 - \|0\|^2) = \|v\|^2 \ge 0$, sehingga \eqref{eq:3.4} terpenuhi.

    \item \textbf{Simetris:}
          Karena $\|v-w\| = \|w-v\|$, maka $\langle v,w \rangle = \langle w,v \rangle$.

    \item \textbf{Kelinieran:}
          Kita akan membuktikan sifat kelinieran berikut:
          \begin{equation}
            \langle \alpha u + \beta v, w \rangle = \alpha \langle u,w \rangle + \beta \langle v,w \rangle. \label{eq:3.5}
          \end{equation}
          Pembuktian ini dilakukan secara bertahap:
          \begin{enumerate}
            \item Sifat negatif:
                  \begin{equation}
                    \langle -v, w \rangle = -\langle v, w \rangle, \label{eq:3.6}
                  \end{equation}
                  berlaku langsung dari definisi. Untuk $v=0$, berakibat $\langle 0,w \rangle = 0$.

            \item Aditivitas: Dari definisi \eqref{eq:3.2} dan identitas jajar genjang \eqref{eq:3.1} (substitusi $x = \frac{u+v}{2} \pm w$, $y = \frac{u-v}{2}$), diperoleh penjabaran:
                  \begin{align*}
                    \langle u,w \rangle + \langle v,w \rangle
                     & = \frac{1}{4} \left[ \left( \|u+w\|^2 + \|v+w\|^2 \right) - \left( \|u-w\|^2 + \|v-w\|^2 \right) \right]                                                                                                         \\
                     & = \frac{1}{4} \left[ \left( 2\left\| \frac{u+v}{2} + w \right\|^2 + 2\left\| \frac{u-v}{2} \right\|^2 \right) - \left( 2\left\| \frac{u+v}{2} - w \right\|^2 + 2\left\| \frac{u-v}{2} \right\|^2 \right) \right] \\
                     & = \frac{1}{4} \left( 2\left\| \frac{u+v}{2} + w \right\|^2 - 2\left\| \frac{u+v}{2} - w \right\|^2 \right)                                                                                                       \\
                     & = 2 \left[ \frac{1}{4} \left( \left\| \frac{u+v}{2} + w \right\|^2 - \left\| \frac{u+v}{2} - w \right\|^2 \right) \right]                                                                                        \\
                     & = 2 \left\langle \frac{1}{2}(u+v), w \right\rangle.
                  \end{align*}
                  Untuk $v=0$, berakibat
                  \begin{equation}
                    \langle u,w \rangle = 2 \left\langle \frac{1}{2}u, w \right\rangle. \label{eq:3.7}
                  \end{equation}
                  Substitusi \eqref{eq:3.7} kembali ke atasnya memberikan sifat \textbf{aditivitas}:
                  \begin{equation}
                    \langle u,w \rangle + \langle v,w \rangle = \left\langle u+v, w \right\rangle. \label{eq:3.8}
                  \end{equation}

            \item Homogenitas: Dari aditivitas \eqref{eq:3.8}, kita peroleh $\langle 2u,w \rangle = \langle u+u, w \rangle = 2\langle u,w \rangle$, dan secara induktif terhadap $m \in \N$ didapat:
                  \begin{equation*}
                    \langle mu, w \rangle = m\langle u,w \rangle.
                  \end{equation*}
                  Dari \eqref{eq:3.7} dan induksi terhadap $n \in \N$ diperoleh:
                  \begin{equation*}
                    \langle u,w \rangle = 2 \left\langle \tfrac{1}{2}u, w \right\rangle \implies \left\langle \tfrac{1}{2}u, w \right\rangle = \tfrac{1}{2}\langle u,w \rangle \implies \cdots \implies \left\langle \tfrac{1}{2^n}u, w \right\rangle = \tfrac{1}{2^n}\langle u,w \rangle.
                  \end{equation*}

                  Kombinasi hasil di atas bersama \eqref{eq:3.6} dan aditivitas \eqref{eq:3.8} membuktikan sifat \textbf{homogenitas}:
                  \begin{equation}
                    \langle \alpha u, w \rangle = \alpha \langle u, w \rangle \label{eq:3.9}
                  \end{equation}
                  berlaku untuk semua bilangan \textit{diadik} $\alpha \in \mathbb{Q}$ yang berbentuk:
                  \begin{equation*}
                    \alpha = m + \sum_{k=1}^n \frac{\alpha_k}{2^k} \quad \text{untuk } m \in \Z,\ n \in \N,\ \alpha_k \in \{0,1\},\ 1 \leq k \leq n.
                  \end{equation*}

                  Selanjutnya, kita buktikan bahwa himpunan bilangan diadik \textbf{padat} (\textit{dense}) di $\R$, yaitu: untuk setiap $x \in \R$ dan $\epsilon > 0$, selalu ada bilangan diadik $\alpha$ dengan $|x - \alpha| < \epsilon$. Konstruksinya:
                  \begin{enumerate}
                    \item Pilih $n \in \N$ cukup besar sehingga $\dfrac{1}{2^n} < \epsilon$.
                    \item Tetapkan $m = \lfloor x \rfloor \in \Z$, sehingga $x - m \in [0,1)$.
                    \item Ekspansikan $x - m$ ke dalam biner tak hingga. Konstruksi $\alpha_k$ dilakukan secara rekursif dengan algoritma \textit{greedy}: definisikan sisa awal $r_0 = x - m \in [0,1)$, lalu untuk tiap $k \geq 1$:
                          \begin{equation*}
                            \alpha_k = \begin{cases} 1 & \text{jika } r_{k-1} \geq \dfrac{1}{2^k}, \\ 0 & \text{jika } r_{k-1} < \dfrac{1}{2^k}, \end{cases} \qquad r_k = r_{k-1} - \frac{\alpha_k}{2^k}.
                          \end{equation*}
                          \textbf{Klaim:} $r_k \in \left[0, \frac{1}{2^k}\right)$ untuk semua $k \geq 0$.

                          \textit{Bukti klaim dengan induksi.} Basis: $r_0 = x - m \in [0,1) = \left[0, \frac{1}{2^0}\right)$, terpenuhi.

                          Langkah induksi: asumsikan $r_{k-1} \in \left[0, \frac{1}{2^{k-1}}\right)$.

                          Jika $r_{k-1} \geq \frac{1}{2^k}$, maka $\alpha_k = 1$ dan $r_k = r_{k-1} - \frac{1}{2^k}$. Karena $r_{k-1} < \frac{1}{2^{k-1}} = \frac{2}{2^k}$, diperoleh
                          \[r_k = r_{k-1} - \frac{1}{2^k} < \frac{2}{2^k} - \frac{1}{2^k} = \frac{1}{2^k}.\]
                          Juga $r_k = r_{k-1} - \frac{1}{2^k} \geq 0$.

                          Jika $r_{k-1} < \frac{1}{2^k}$, maka $\alpha_k = 0$ dan $r_k = r_{k-1} \in \left[0, \frac{1}{2^k}\right)$. Pada kedua kasus, $r_k \in \left[0, \frac{1}{2^k}\right)$, sehingga $r_k \to 0$ untuk $k \to \infty$. Karena $r_0 = \sum_{j=1}^{k} \frac{\alpha_j}{2^j} + r_k$ untuk setiap $k$, maka:
                          \begin{equation*}
                            x - m = r_0 = \lim_{k \to \infty} \sum_{j=1}^{k} \frac{\alpha_j}{2^j} = \sum_{k=1}^{\infty} \frac{\alpha_k}{2^k}, \quad \alpha_k \in \{0,1\}.
                          \end{equation*}
                    \item Definisikan bilangan diadik $\alpha$ sebagai pemotongan di suku ke-$n$:
                          \begin{equation*}
                            \alpha = m + \sum_{k=1}^n \frac{\alpha_k}{2^k}.
                          \end{equation*}
                  \end{enumerate}

                  Maka jarak antara $x$ dan $\alpha$ adalah sisa deret:
                  \begin{equation*}
                    |x - \alpha| = \sum_{k=n+1}^{\infty} \frac{\alpha_k}{2^k} \leq \sum_{k=n+1}^{\infty} \frac{1}{2^k} = \frac{1}{2^n} < \epsilon.
                  \end{equation*}
                  Jadi, setiap bilangan real merupakan limit suatu barisan diadik. Dengan kekontinuan norma, homogenitas \eqref{eq:3.9} meluas ke semua $\alpha \in \R$.
          \end{enumerate}
  \end{itemize}
  Dengan gabungan aditivitas dan homogenitas, sifat kelinieran \eqref{eq:3.5} terpenuhi atas $\R$.
\end{frame}

\begin{frame}[allowframebreaks]{\insertsection}{\insertsubsection}
  \textbf{Kasus ruang linear atas $\C$}: \\
  Definisikan $\langle \cdot, \cdot \rangle : \mathcal{F} \times \mathcal{F} \to \C$ lewat identitas polarisasi kompleks \eqref{eq:3.3}. Misalkan $\langle \cdot, \cdot \rangle_R$ adalah bagian realnya, sehingga:
  \begin{equation*}
    \langle v,w \rangle = \langle v,w \rangle_R + i\langle v,iw \rangle_R.
  \end{equation*}
  \begin{itemize}
    \item \textbf{Definit Positif:} Langsung dari definisi,
          \begin{equation*}
            \langle v,v \rangle = \langle v,v \rangle_R + i\langle v,iv \rangle_R = \|v\|^2 + i \cdot 0 = \|v\|^2 \geq 0.
          \end{equation*}

    \item \textbf{Aditivitas:} Diwarisi langsung dari linearitas bagian realnya:
          \begin{equation*}
            \langle u+v, w \rangle_R = \langle u,w \rangle_R + \langle v,w \rangle_R,
          \end{equation*}
          dan begitu pula untuk suku imajinernya, sehingga $\langle u+v, w \rangle = \langle u,w \rangle + \langle v,w \rangle$.

    \item \textbf{Homogenitas skalar kompleks:} Cukup diverifikasi untuk perkalian dengan $i$ (kasus $\alpha = i$), sebab skalar riil sudah terbukti di atas. Dari definisi:
          \begin{align*}
            \langle iv, w \rangle_R & = \frac{1}{4}\left(\|iv+w\|^2 - \|iv-w\|^2\right).
          \end{align*}
          Perhatikan bahwa $\|iv \pm w\| = \|v \mp iw\|$ karena $\|i \cdot z\| = \|z\|$. Maka:
          \begin{align*}
            \langle iv, w \rangle_R  & = \frac{1}{4}\left(\|w - iv\|^2 - \|w + iv\|^2\right) = -\langle v, iw \rangle_R,                                             \\
            \langle iv, iw \rangle_R & = \frac{1}{4}\left(\|iv+iw\|^2 - \|iv-iw\|^2\right) = \frac{1}{4}\left(\|v+w\|^2 - \|v-w\|^2\right) = \langle v, w \rangle_R.
          \end{align*}

          Dengan demikian:
          \begin{align*}
            \langle iv, w \rangle & = \langle iv, w \rangle_R + i\langle iv, iw \rangle_R                                      \\
                                  & = -\langle v, iw \rangle_R + i\langle v, w \rangle_R                                       \\
                                  & = i\left(\langle v, w \rangle_R + i\langle v, iw \rangle_R\right) = i\langle v, w \rangle.
          \end{align*}
          Karena homogenitas riil dan perkalian $i$ keduanya terbukti, maka $\langle \alpha v, w \rangle = \alpha\langle v, w \rangle$ untuk semua $\alpha \in \C$.

    \item \textbf{Simetris Konjugat:} Perlu ditunjukkan $\langle w, v \rangle = \overline{\langle v, w \rangle}$. Dari bagian realnya:
          \begin{equation*}
            \langle w, v \rangle_R = \frac{1}{4}\left(\|w+v\|^2 - \|w-v\|^2\right) = \langle v, w \rangle_R,
          \end{equation*}
          karena $\|w-v\| = \|v-w\|$. Untuk bagian imajinernya, karena $\langle w, iw \rangle_R$ dan $\langle w,iv \rangle_R = \langle v, iw \rangle_R$ (serupa), kita peroleh:
          \begin{equation*}
            \langle w, v \rangle = \langle w, v \rangle_R + i\langle w, iv \rangle_R = \langle v, w \rangle_R - i\langle v, iw \rangle_R = \overline{\langle v, w \rangle}.
          \end{equation*}
  \end{itemize}
  Dengan demikian, seluruh aksioma hasil kali dalam atas $\C$ terpenuhi.
\end{frame}

\subsection{Akibat 3.11}

\begin{frame}[allowframebreaks]{\insertsection}{\insertsubsection}
  \setcounter{akibat}{10}
  \begin{akibat}
    Setiap hasil kali dalam adalah kontinu.
  \end{akibat}
  \textit{Bukti.}
  Misalkan $(v_n) \to v$ dan $(w_m) \to w$ di $\mathcal{F}$. Karena $v_n \to v$ dan $w_m \to w$, maka $v_n \pm w_m \to v \pm w$, dan berdasarkan kekontinuan norma $\|\cdot\|$:
  \begin{equation*}
    \|v_n \pm w_m\| \to \|v \pm w\| \quad \text{untuk } n, m \to \infty.
  \end{equation*}

  \textbf{Kasus $\mathcal{F}$ atas $\R$:} Dengan identitas polarisasi \eqref{eq:3.2}:
  \begin{align*}
    \langle v_n, w_m \rangle & = \frac{1}{4}\left(\|v_n+w_m\|^2 - \|v_n-w_m\|^2\right)
    \xrightarrow{n,m\to\infty} \frac{1}{4}\left(\|v+w\|^2 - \|v-w\|^2\right) = \langle v, w \rangle.
  \end{align*}

  \textbf{Kasus $\mathcal{F}$ atas $\C$:} Tambahan lagi $v_n \pm iw_m \to v \pm iw$, sehingga dengan identitas polarisasi kompleks \eqref{eq:3.3}:
  \begin{align*}
    \langle v_n, w_m \rangle & = \frac{1}{4}\left(\|v_n+w_m\|^2 - \|v_n-w_m\|^2 + i\|v_n+iw_m\|^2 - i\|v_n-iw_m\|^2\right)                                  \\
                             & \xrightarrow{n,m\to\infty} \frac{1}{4}\left(\|v+w\|^2 - \|v-w\|^2 + i\|v+iw\|^2 - i\|v-iw\|^2\right) = \langle v, w \rangle.
  \end{align*}

  Pada kedua kasus, $\langle v_n, w_m \rangle \to \langle v, w \rangle$, sehingga $\langle \cdot, \cdot \rangle$ kontinu. \qed
\end{frame}

\begin{frame}[allowframebreaks]{\insertsection}
  \setcounter{definisi}{11}
  \begin{definisi}
    Sebuah ruang Euclidean yang lengkap disebut \textbf{ruang Hilbert}.
  \end{definisi}
  \begin{definisi}
    Sebuah himpunan bagian tak-kosong $\mathcal{K} \subset \mathcal{F}$ disebut \textbf{konveks}, jika untuk sebarang $u,v \in \mathcal{K}$ segmen garis
    \begin{equation*}
      [u,v] = \{ \lambda u + (1-\lambda)v \mid \lambda \in [0,1] \}
    \end{equation*}
    antara $u$ dan $v$ terletak di dalam $\mathcal{K}$, yaitu, jika $[u,v] \subset \mathcal{K}$ untuk semua $u,v \in \mathcal{K}$.
    Jika untuk sebarang $u,v \in \mathcal{K}$ dengan $u \neq v$, segmen garis terbuka
    \begin{equation*}
      (u,v) = \{ \lambda u + (1-\lambda)v \mid \lambda \in (0,1) \}
    \end{equation*}
    terkandung di dalam interior dari $\mathcal{K}$, maka $\mathcal{K}$ disebut \textbf{konveks tegas}.
  \end{definisi}
\end{frame}

\subsection{Teorema 3.14}

\begin{frame}[allowframebreaks]{\insertsection}{\insertsubsection}
  \setcounter{teorema}{13}
  \begin{teorema}
    Misalkan $\mathcal{F}$ adalah sebuah ruang Hilbert dengan hasil kali dalam $\langle \cdot, \cdot \rangle$ dan norma $\| \cdot \| = \langle \cdot, \cdot \rangle^{1/2}$. Selanjutnya, misalkan $\mathcal{S} \subset \mathcal{F}$ menjadi sebuah himpunan bagian tertutup dan konveks dari $\mathcal{F}$. Maka terdapat untuk sebarang $f \in \mathcal{F}$ sebuah aproksimasi terbaik $s^* \in \mathcal{S}$ terhadap $f$.
  \end{teorema}
  \textit{Bukti.}
  Definisikan jarak minimal $\eta \equiv \eta(f, \mathcal{S}) := \inf_{s \in \mathcal{S}} \|s - f\|$. Kita terlebih dahulu membuktikan keberadaan barisan minimal di $\mathcal{S}$.

  Karena $\mathcal{S}$ tak-kosong sesuai definisi himpunan konveks yang digunakan, ambil $s_0\in\mathcal{S}$. Semua jarak tak-negatif, sehingga:
  \begin{equation*}
    0\leq\eta\leq\|s_0-f\|<\infty.
  \end{equation*}
  Untuk setiap $n\in\N$, $n\geq 1$, berlaku $\eta+1/n>\eta$. Berdasarkan sifat infimum, harus ada $s_n\in\mathcal{S}$ dengan $\|s_n-f\|<\eta+1/n$. Sebab, jika tidak ada, maka $\|s-f\|\geq\eta+1/n$ untuk semua $s\in\mathcal{S}$. Ini menjadikan $\eta+1/n$ suatu batas bawah yang lebih besar daripada $\eta$, bertentangan dengan definisi infimum sebagai batas bawah terbesar.

  Dengan memilih satu elemen tersebut untuk setiap $n$, diperoleh barisan $(s_n)_{n\in\N}\subset\mathcal{S}$ yang memenuhi:
  \begin{equation*}
    \eta\leq\|s_n-f\|<\eta+\frac{1}{n},
    \qquad 0\leq\|s_n-f\|-\eta<\frac{1}{n}.
  \end{equation*}
  Karena $1/n\to 0$, teorema apit memberikan $\|s_n-f\|\to\eta$. Barisan inilah yang disebut \textbf{barisan minimal}. Selanjutnya, kita buktikan bahwa barisan tersebut konvergen ke suatu aproksimasi terbaik.

  Terapkan identitas jajar genjang \eqref{eq:3.1} dengan substitusi $v = s_n - f$ dan $w = s_m - f$, sehingga:
  \begin{equation*}
    v - w = s_n - s_m, \qquad v + w = s_n + s_m - 2f = 2\left(\tfrac{s_n+s_m}{2} - f\right).
  \end{equation*}
  Diperoleh:
  \begin{equation*}
    \|s_n - s_m\|^2 = 2\|s_n - f\|^2 + 2\|s_m - f\|^2 - 4\left\|\frac{s_n+s_m}{2} - f\right\|^2.
  \end{equation*}

  Karena $\mathcal{S}$ konveks dan $s_n, s_m \in \mathcal{S}$, maka $\frac{s_n+s_m}{2} \in \mathcal{S}$, sehingga $\left\|\frac{s_n+s_m}{2} - f\right\| \geq \eta$. Maka:
  \begin{equation*}
    \|s_n - s_m\|^2 \leq 2\|s_n - f\|^2 + 2\|s_m - f\|^2 - 4\eta^2.
  \end{equation*}

  Karena $\|s_n - f\| \to \eta$ dan $\|s_m - f\| \to \eta$, ruas kanan konvergen ke $2\eta^2 + 2\eta^2 - 4\eta^2 = 0$ saat $n, m \to \infty$. Jadi untuk setiap $\varepsilon > 0$ terdapat $N = N(\varepsilon) \in \N$ sedemikian sehingga:
  \begin{equation*}
    \|s_n - s_m\| < \varepsilon \quad \text{untuk semua } n, m \geq N.
  \end{equation*}
  Ini berarti $(s_n)$ adalah barisan Cauchy. Karena $\mathcal{F}$ adalah ruang Hilbert (lengkap), barisan ini konvergen ke suatu limit $s^* \in \mathcal{F}$.

  Karena $\mathcal{S}$ tertutup, maka $s^* \in \mathcal{S}$. Kekontinuan norma memberikan:
  \begin{equation*}
    \|s^* - f\| = \left\|\lim_{n \to \infty} s_n - f\right\| = \lim_{n \to \infty} \|s_n - f\| = \eta.
  \end{equation*}
  Jadi $s^* \in \mathcal{S}$ mencapai infimum $\eta$, yang berarti $s^*$ adalah aproksimasi terbaik dari $f$ di $\mathcal{S}$. \qed

\end{frame}
